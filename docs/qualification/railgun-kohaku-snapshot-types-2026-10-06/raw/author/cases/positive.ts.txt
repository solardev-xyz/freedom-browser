import {
  createRailgunKohakuSnapshotPlugin,
  type SnapshotHost,
  type SnapshotReadPlugin,
  type PinnedReadTarget,
  type ReadNote,
  type ReadAmount,
  type SnapshotExtras,
} from 'freedom-snapshot-contract';

const note: ReadNote = {
  id: '0:0', tree: 0, position: 0,
  txid: '0x01', hash: '0x02', tokenHash: '0x03',
  asset: { __type: 'erc20', contract: '0x04' },
  amount: 1n, tag: 'unverified', spentTxid: false,
};
// Deliberately type-level examples; runtime hash/address/length bounds are
// separately enforced by the adapter and are not implied by template types.
const host: SnapshotHost = {
  signal: new AbortController().signal,
  capture() {
    return {
      snapshot: { instanceId: 'example', received: [note] },
      assertCurrent() { return undefined; },
    };
  },
};
const adapter: SnapshotReadPlugin = createRailgunKohakuSnapshotPlugin({
  host, signal: new AbortController().signal,
});
const target: PinnedReadTarget = adapter;
const factory: (options: { host: SnapshotHost; signal: AbortSignal }) => SnapshotReadPlugin = createRailgunKohakuSnapshotPlugin;
type NoPrepare<T> = Extract<keyof T, `prepare${string}`> extends never ? true : false;
const noAdapterPrepare: NoPrepare<SnapshotReadPlugin> = true;
const noTargetPrepare: NoPrepare<PinnedReadTarget> = true;
async function consume() {
  const id: string = await target.instanceId();
  const balances = await target.balance(undefined);
  balances.push({ asset: { __type: 'native' }, amount: 2n, tag: 'unverified' });
  const notes = await adapter.notes(undefined, true);
  notes.push(note);
  for (const row of notes) row.asset = { __type: 'erc721', contract: '0x05', tokenId: 3n };
  const provenance: 'host-supplied' = target.provenance;
  // Upstream notes is optional; the adapter's specialized notes is required.
  if (target.notes) {
    const pinnedNotes: ReadNote[] = await target.notes();
    pinnedNotes.push(note);
  }
  return { id, provenance, balances, notes };
}
export { target, factory, consume, noAdapterPrepare, noTargetPrepare };

import type { PICapCfg, PICapabilities, PluginInstance } from 'kohaku-plugin-base';
type Caps = {
  features: {};
  note: ReadNote;
  assetAmounts: { input: ReadAmount; internal: ReadAmount; output: ReadAmount; read: ReadAmount };
  extras: SnapshotExtras;
};
const capabilitiesValid: PICapCfg<Caps> extends PICapabilities ? true : false = true;
const actualUpstreamInstance: PluginInstance<string, Caps> = target;
export { capabilitiesValid, actualUpstreamInstance };

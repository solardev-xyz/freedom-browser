/** Emitted adapter behavior with independent trusted hosts, not Freedom authority. */
const assert = require('assert/strict');
const pub = require('./public-host.cjs');
const priv = require('./independent-host.cjs');
const PUBLIC = 'RAILGUN_KOHAKU_PUBLIC_ADAPTER_REFUSED';
const PRIVATE = 'RAILGUN_KOHAKU_PRIVATE_ADAPTER_REFUSED';
const CONTRACT = 'RAILGUN_KOHAKU_PUBLIC_ADAPTER_CONTRACT';
const tick = () => new Promise((resolve) => setImmediate(resolve));
async function bounded(promise, label) {
  let timer;
  try {
    return await Promise.race([
      promise,
      new Promise((_, reject) => {
        timer = setTimeout(() => reject(Error(label)), 3000);
      }),
    ]);
  } finally {
    clearTimeout(timer);
  }
}
function unobservable(kind) {
  const held = pub.deferred();
  let entries = 0;
  const descriptor = {
    get() {
      entries++;
      throw Error('unobservable host work');
    },
  };
  if (kind === 'constructor') Object.defineProperty(held.promise, 'constructor', descriptor);
  else
    Object.defineProperty(held.promise, 'constructor', {
      value: Object.defineProperty({}, Symbol.species, descriptor),
    });
  return { held, entries: () => entries };
}
async function runMatrix(api, other, clean, crossEntry) {
  const cases = [],
    unhandled = [];
  const onUnhandled = (reason) => unhandled.push(reason);
  process.on('unhandledRejection', onUnhandled);
  async function test(name, action) {
    const opened = [],
      releases = [];
    function open(lane = 'public', change = () => {}, selected = api) {
      const fixture =
        lane === 'public' ? pub.independentPublicHost() : priv.independentPrivateHost();
      change(fixture);
      const create =
        lane === 'public'
          ? selected.createRailgunKohakuPublicAdapter
          : selected.createRailgunKohakuPrivateAdapter;
      const adapter = create({ host: fixture.host, signal: new AbortController().signal });
      // Observe rejection immediately; assertions below retain its actual state.
      adapter.closed.catch(() => {});
      opened.push(adapter);
      const sender =
        lane === 'public'
          ? selected.createRailgunKohakuPublicAdapterSubmitter(adapter)
          : selected.createRailgunKohakuPrivateAdapterBroadcaster(adapter);
      return {
        ...fixture,
        adapter,
        sender,
        prepare: () =>
          lane === 'public'
            ? adapter.prepareShield(fixture.input)
            : adapter.prepareTransfer(fixture.input, priv.INSTANCE),
        send: (token) => (lane === 'public' ? sender.submit(token) : sender.broadcast(token)),
        sends: () =>
          fixture.calls.filter((c) =>
            lane === 'public' ? c === 'submit' : c.method === 'broadcast'
          ).length,
        closes: () =>
          fixture.calls.filter((c) => (lane === 'public' ? c === 'close' : c.method === 'close'))
            .length,
      };
    }
    try {
      await bounded(action(open, releases), 'Case timeout: ' + name);
      await tick();
      assert.deepEqual(unhandled, []);
      clean();
      cases.push(name);
    } finally {
      for (const release of releases) release();
      for (const adapter of opened) adapter.close();
      await bounded(Promise.allSettled(opened.map((a) => a.closed)), 'Cleanup timeout: ' + name);
    }
  }
  try {
    await test('public-baseline', async () => {
      assert.deepEqual(
        await pub.checkPublicConformance(
          api.createRailgunKohakuPublicAdapter,
          api.createRailgunKohakuPublicAdapterSubmitter
        ),
        { reads: 4, preparation: 1, submission: 1, close: 1, authority: false }
      );
    });
    for (const code of ['PRIVATE_BROADCAST_UNCERTAIN', 'RAILGUN_KOHAKU_REFUSED'])
      await test('public-original-error:' + code, async (open) => {
        const error = Object.assign(
          Error('independent host reason'),
          { code },
          code === 'PRIVATE_BROADCAST_UNCERTAIN'
            ? { transactionHash: pub.HASH, submissionStatus: 'unknown' }
            : {}
        );
        const f = open('public', (f) => {
          f.host.submit = () => Promise.reject(error);
        });
        const token = await f.prepare();
        await assert.rejects(f.send(token), (reason) => reason === error);
        await f.adapter.closed;
        await assert.rejects(f.send(token), { code: PUBLIC });
      });
    await test('public-detached-amount-and-token-replay', async (open) => {
      const f = open(),
        token = await f.prepare();
      f.input.amount = 1001n;
      f.input.asset.__type = 'erc20';
      await assert.rejects(f.send({ ...token }), { code: PUBLIC });
      const result = await f.send(token);
      assert.equal(result, f.acknowledged);
      assert.equal(result.value, '1000');
      await f.adapter.closed;
      await assert.rejects(f.send(token), { code: PUBLIC });
      assert.equal(f.sends(), 1);
      assert.equal(f.closes(), 1);
    });
    await test('public-wrong-acknowledged-amount', async (open) => {
      const f = open(),
        token = await f.prepare();
      f.acknowledged.value = '1001';
      await assert.rejects(f.send(token), { code: CONTRACT, submissionMayHaveOccurred: true });
      await f.adapter.closed;
      await assert.rejects(f.send(token), { code: PUBLIC });
      assert.equal(f.sends(), 1);
    });
    const addresses = [
      ['lower', '0x8ba1f109551bd432803012645ac136ddd64dba72', true],
      ['checksum', '0x8ba1f109551bD432803012645Ac136ddd64DBA72', true],
      ['unprefixed', '8ba1f109551bd432803012645ac136ddd64dba72', false],
      ['icap', 'XE65GB6LDNXYOFTX0NSV3FUWKOWIXAMJK36', false],
      ['bad-checksum', '0x8Ba1f109551bD432803012645Ac136ddd64DBA72', false],
    ];
    for (const lane of ['public', 'private'])
      for (const field of ['from', 'to'])
        for (const [name, address, valid] of addresses)
          await test(lane + '-address:' + field + ':' + name, async (open) => {
            const f = open(lane),
              token = await f.prepare();
            f.acknowledged[field] = address;
            if (valid) {
              assert.equal(await f.send(token), f.acknowledged);
              assert.equal(f.acknowledged[field], address);
            } else if (lane === 'public')
              await assert.rejects(f.send(token), {
                code: CONTRACT,
                submissionMayHaveOccurred: true,
              });
            else
              assert.deepEqual(await f.send(token), {
                transactionHash: priv.HASH,
                submissionStatus: 'unknown',
              });
            await f.adapter.closed;
            assert.equal(f.sends(), 1);
          });
    await test('public-private-cross-kind-refusal-preserves-own-token', async (open) => {
      const a = open(),
        b = open('private'),
        [at, bt] = await Promise.all([a.prepare(), b.prepare()]);
      assert.throws(() => api.createRailgunKohakuPublicAdapterSubmitter(b.adapter), {
        code: PUBLIC,
      });
      assert.throws(() => api.createRailgunKohakuPrivateAdapterBroadcaster(a.adapter), {
        code: PRIVATE,
      });
      await assert.rejects(a.send(bt), { code: PUBLIC });
      await assert.rejects(b.send(at), { code: PRIVATE });
      assert.equal(a.sends(), 0);
      assert.equal(b.sends(), 0);
      assert.equal(await a.send(at), a.acknowledged);
      assert.equal(await b.send(bt), b.acknowledged);
      await Promise.all([a.adapter.closed, b.adapter.closed]);
    });
    await test('public-cross-entry-same-registry', async (open) => {
      const f = open();
      assert.throws(
        () =>
          crossEntry.createRailgunKohakuPublicAdapter({
            host: f.host,
            signal: new AbortController().signal,
          }),
        { code: PUBLIC }
      );
      const token = await f.prepare();
      const sender = crossEntry.createRailgunKohakuPublicAdapterSubmitter(f.adapter);
      assert.equal(await sender.submit(token), f.acknowledged);
      await f.adapter.closed;
      await assert.rejects(sender.submit(token), { code: PUBLIC });
      assert.equal(f.sends(), 1);
    });
    for (const lane of ['public', 'private'])
      for (const kind of ['constructor', 'species'])
        for (const operation of ['read', 'prepare', 'submit', 'close'])
          await test(lane + '-unobservable:' + kind + ':' + operation, async (open, releases) => {
            const bad = unobservable(kind);
            releases.push(() => bad.held.resolve());
            const code = lane === 'public' ? PUBLIC : PRIVATE;
            const method =
              operation === 'read'
                ? 'notes'
                : operation === 'prepare'
                  ? lane === 'public'
                    ? 'prepareShield'
                    : 'prepareTransfer'
                  : operation === 'submit'
                    ? lane === 'public'
                      ? 'submit'
                      : 'broadcast'
                    : 'close';
            const f = open(lane, (f) => {
              const close = f.host.close;
              f.host[method] =
                operation === 'close'
                  ? () => {
                      close();
                      return bad.held.promise;
                    }
                  : () => bad.held.promise;
            });
            if (operation === 'close')
              assert.equal(await f.send(await f.prepare()), f.acknowledged);
            else {
              const pending =
                operation === 'read'
                  ? f.adapter.notes()
                  : operation === 'prepare'
                    ? f.prepare()
                    : f.send(await f.prepare());
              await assert.rejects(
                pending,
                lane === 'public' && operation === 'submit'
                  ? { code: CONTRACT, submissionMayHaveOccurred: true }
                  : { code }
              );
            }
            assert.equal(bad.entries(), 1);
            assert.equal(f.adapter.signal.aborted, true);
            assert.equal(f.closes(), 1);
            await assert.rejects(f.adapter.closed, { code });
            await assert.rejects(f.prepare(), { code });
            bad.held.resolve();
            await tick();
            await assert.rejects(f.adapter.closed, { code });
          });
    await test('public-decorated-pending-retains-logical-drain', async (open, releases) => {
      const raw = pub.deferred();
      raw.promise.extra = true;
      releases.push(() => raw.resolve());
      const f = open('public', (f) => {
        f.host.submit = () => raw.promise;
      });
      let closed = false;
      f.adapter.closed.then(() => {
        closed = true;
      });
      await assert.rejects(f.send(await f.prepare()), {
        code: CONTRACT,
        submissionMayHaveOccurred: true,
      });
      await tick();
      assert.equal(closed, false);
      raw.resolve();
      await f.adapter.closed;
      assert.equal(closed, true);
    });
    await test('public-original-held-rejection-before-host-drain', async (open, releases) => {
      const held = pub.deferred(),
        drain = pub.deferred(),
        error = Object.assign(Error('cancelled'), { code: 'RAILGUN_KOHAKU_REFUSED' });
      releases.push(
        () => held.reject(error),
        () => drain.resolve()
      );
      const f = open('public', (f) => {
        f.host.submit = () => held.promise;
        f.host.closed = drain.promise;
        f.host.close = () => {
          f.calls.push('close');
        };
      });
      const token = await f.prepare(),
        pending = f.send(token);
      let closed = false;
      f.adapter.closed.then(() => {
        closed = true;
      });
      f.adapter.close();
      held.reject(error);
      await assert.rejects(pending, (e) => e === error);
      await tick();
      assert.equal(closed, false);
      await assert.rejects(f.send(token), { code: PUBLIC });
      drain.resolve();
      await f.adapter.closed;
      assert.equal(f.closes(), 1);
    });
    if (other)
      for (const lane of ['public', 'private'])
        await test(lane + '-physical-copy-isolation-distinct-hosts', async (open) => {
          const a = open(lane),
            b = open(lane, () => {}, other),
            [one, two] = await Promise.all([a.prepare(), b.prepare()]);
          const code = lane === 'public' ? PUBLIC : PRIVATE;
          const factory =
            lane === 'public'
              ? 'createRailgunKohakuPublicAdapterSubmitter'
              : 'createRailgunKohakuPrivateAdapterBroadcaster';
          assert.throws(() => other[factory](a.adapter), { code });
          await assert.rejects(a.send(two), { code });
          await assert.rejects(b.send(one), { code });
          assert.equal(a.sends(), 0);
          assert.equal(b.sends(), 0);
          assert.equal(await a.send(one), a.acknowledged);
          assert.equal(await b.send(two), b.acknowledged);
          await Promise.all([a.adapter.closed, b.adapter.closed]);
        });
    return {
      cases,
      count: cases.length,
      unhandledRejections: unhandled.length,
      sharedHostAcrossCopiesSupported: false,
      foreignHostAuthorityQualified: false,
    };
  } finally {
    process.removeListener('unhandledRejection', onUnhandled);
  }
}
module.exports = { runMatrix };

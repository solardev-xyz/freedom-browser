"""Verify exact artifact copies and completed runtime consumer evidence, read-only inputs."""
import hashlib
import json
from pathlib import Path

if not __debug__:
    raise RuntimeError('Optimized Python is unsupported')
root = Path(__file__).resolve().parent
lane = Path('<RUNTIME_BUILD>')

def sha(path):
    assert path.is_file() and not path.is_symlink()
    return hashlib.sha256(path.read_bytes()).hexdigest()

def inventory(directory):
    result = {}
    for path in sorted(directory.rglob('*')):
        assert not path.is_symlink()
        if path.is_file():
            result[str(path.relative_to(directory))] = sha(path)
        else:
            assert path.is_dir()
    return result

prepared = json.loads((root / 'PREPARED.json').read_text())
assert sha(lane / 'FREEZE.json') == prepared['artifactFreezeSha256']
for name,digest in json.loads((lane / 'FREEZE.json').read_text())['files'].items():
    assert sha(lane / name) == digest
assert inventory(lane / 'build-a/artifact') == prepared['artifactInventory']
assert inventory(lane / 'build-b/artifact') == prepared['artifactInventory']
rows = []
traps = {'trap-network': 'global.fetch', 'trap-process': 'module:child_process', 'trap-root': 'outside-artifact-module'}
for mode in prepared['modes']:
    directory = root / ('consumer-' + mode)
    namespace = directory / 'node_modules/@freedom'
    assert inventory(namespace / 'restricted-railgun-node-prototype') == prepared['artifactInventory']
    if mode == 'mixed':
        assert inventory(namespace / 'restricted-railgun-node-copy') == prepared['artifactInventory']
    for name,digest in prepared['consumerInputs'].items():
        assert sha(directory / name) == digest == sha(root / name)
    row = {'mode': mode, 'logSha256': sha(directory / 'consumer.log')}
    if mode in traps:
        assert not (directory / 'report.json').exists()
        log = (directory / 'consumer.log').read_text()
        assert 'Sticky forbidden work' in log and traps[mode] in log
        row['expectedTrap'] = traps[mode]
    else:
        report = json.loads((directory / 'report.json').read_text())
        assert report['mode'] == mode
        assert report['node'] == 'v24.18.1'
        assert report['stickyForbiddenWork'] == []
        assert report['matrix']['unhandledRejections'] == 0
        assert report['matrix']['count'] == (47 if mode == 'mixed' else 45)
        assert report['snapshotReads'] == 4
        assert len(report['privateOperations']) == 3
        assert report['hostAuthorityQualified'] is False
        assert report['genuineFreedomHostIncluded'] is False
        assert report['osEgressQualified'] is False
        assert report['matrix']['sharedHostAcrossCopiesSupported'] is False
        assert report['sharedCjsEsmIdentity'] is (True if mode == 'mixed' else None)
        if mode == 'mixed':
            assert sum('physical-copy-isolation-distinct-hosts' in name for name in report['matrix']['cases']) == 2
        row.update(reportSha256=sha(directory / 'report.json'),pid=report['pid'],matrixCases=report['matrix']['count'])
    rows.append(row)
assert len({row['pid'] for row in rows if 'pid' in row}) == 3
(root / 'VERIFIED-RESULTS.json').write_text(json.dumps({
    'artifactFreezeSha256': prepared['artifactFreezeSha256'],
    'allOriginalBuildFreezeEntriesUnchanged': True,
    'allSevenPhysicalPackageCopiesUnchanged': True,
    'allConsumerInputsUnchanged': True,
    'positiveRuntimeModes': 3,
    'matrixCaseExecutions': 137,
    'expectedTrapFailures': 3,
    'scope': 'Result/report verification; actual original command exit observations are separate',
    'cases': rows,
},indent=2)+'\n')
print(json.dumps({'verifiedModes':6,'matrixCaseExecutions':137,'allArtifactBytesUnchanged':True}))

import cjs = require('@freedom/restricted-railgun-node-prototype');
import {createRailgunKohakuPrivateAdapter, createRailgunKohakuPrivateAdapterBroadcaster, type PrivateOperation, type PrivateSubmissionOutcome} from '@freedom/restricted-railgun-node-prototype';
declare const host: cjs.RestrictedPrivateHost;
declare const signal: AbortSignal;
declare const input: cjs.PrivateInput;
const cjsAdapter = cjs.createRailgunKohakuPrivateAdapter({host,signal});
const esmAdapter = createRailgunKohakuPrivateAdapter({host,signal});
async function exchange() {
  const fromCjs = await cjsAdapter.prepareTransfer(input, 'recipient');
  const esmOperation: PrivateOperation = fromCjs;
  const fromEsm = await esmAdapter.prepareTransfer(input, 'recipient');
  const cjsOperation: cjs.PrivateOperation = fromEsm;
  const esmOutcome: PrivateSubmissionOutcome = await createRailgunKohakuPrivateAdapterBroadcaster(cjsAdapter).broadcast(esmOperation);
  const cjsOutcome: cjs.PrivateSubmissionOutcome = await cjs.createRailgunKohakuPrivateAdapterBroadcaster(esmAdapter).broadcast(cjsOperation);
  return {esmOutcome,cjsOutcome};
}
export {exchange};

import {createRailgunKohakuPublicAdapter,createRailgunKohakuPublicAdapterSubmitter,type PublicShieldOperation,type PublicShieldAcknowledgement} from '@freedom/restricted-railgun-node-prototype';
declare const publicHost: cjs.RestrictedPublicHost;
declare const publicInput: cjs.PublicShieldInput;
const publicCjs = cjs.createRailgunKohakuPublicAdapter({host:publicHost,signal});
const publicEsm = createRailgunKohakuPublicAdapter({host:publicHost,signal});
async function publicExchange() {
 const fromCjs = await publicCjs.prepareShield(publicInput);
 const esmOperation: PublicShieldOperation = fromCjs;
 const fromEsm = await publicEsm.prepareShield(publicInput);
 const cjsOperation: cjs.PublicShieldOperation = fromEsm;
 const esmAck: PublicShieldAcknowledgement = await createRailgunKohakuPublicAdapterSubmitter(publicCjs).submit(esmOperation);
 const cjsAck: cjs.PublicShieldAcknowledgement = await cjs.createRailgunKohakuPublicAdapterSubmitter(publicEsm).submit(cjsOperation);
 const copied: PublicShieldOperation = {...esmOperation};
 return {esmAck,cjsAck,copied};
}
export {publicExchange};

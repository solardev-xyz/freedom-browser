"""Read-only comparison of two fresh builder outputs; no runtime loading."""
import hashlib
import json
from pathlib import Path

if not __debug__:
    raise RuntimeError('Optimized Python is unsupported')
root = Path(__file__).resolve().parent

def inventory(base):
    result = {}
    for path in sorted(base.rglob('*')):
        assert not path.is_symlink()
        if path.is_file():
            result[str(path.relative_to(base))] = hashlib.sha256(path.read_bytes()).hexdigest()
        else:
            assert path.is_dir()
    return result

first = inventory(root / 'build-a')
second = inventory(root / 'build-b')
assert first == second
artifact = {name[len('artifact/'):]: value for name, value in first.items() if name.startswith('artifact/')}
assert len(artifact) == 21
report = json.loads((root / 'build-a/BUILD-REPORT.json').read_text())
assert report['runtimeConsumerQualification'] is False
assert report['declarationQualification'] is False
assert report['fixedFreedomHostsIncluded'] is False
assert report['bridgeResolutions'] == {
    'src/main/wallet/railgun-kohaku-private-adapter.js': 1,
    'src/main/wallet/railgun-kohaku-public-adapter.js': 1,
}
assert {item['path'] for item in report['externalImports']} == {'assert/strict', 'util'}
assert len(report['inputInventory']) == 82
for name in ('build-a', 'build-b'):
    assert 'types' not in json.loads((root / name / 'artifact/package.json').read_text())
(root / 'ARTIFACT-INVENTORY.json').write_text(json.dumps(artifact, indent=2) + '\n')
(root / 'REPRODUCIBILITY.json').write_text(json.dumps({
    'outputsCompared': ['build-a', 'build-b'],
    'allOutputFilesByteIdentical': True,
    'allOutputFiles': len(first),
    'artifactFiles': len(artifact),
    'hashes': first,
    'parsedInputs': report['parsedInputs'],
    'emittedInputs': len(report['emittedInputs']),
    'outputBytes': report['outputBytes'],
    'inputInventoryEntries': len(report['inputInventory']),
    'runtimeConsumersExecuted': False,
    'compilerExecuted': False,
}, indent=2) + '\n')
print(json.dumps({'allOutputFilesByteIdentical': True, 'artifactFiles': len(artifact), 'allOutputFiles': len(first)}))

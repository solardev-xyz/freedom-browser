def sha(path):
    assert path.is_file() and not path.is_symlink()
    return hashlib.sha256(path.read_bytes()).hexdigest()

def inventory(base):
    result = {}
    for path in sorted(base.rglob('*')):
        assert not path.is_symlink()
        if path.is_file():
            result[str(path.relative_to(base))] = sha(path)
        else:
            assert path.is_dir()
    return result

import hashlib,json,pathlib,shutil,sys
base=pathlib.Path(__file__).parent
artifact=pathlib.Path(sys.argv[1]);inventory_path=pathlib.Path(sys.argv[2]);expected_inventory_sha=sys.argv[3]
source=pathlib.Path('<HISTORICAL_TYPES>')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(inventory_path)==expected_inventory_sha
assert sha(source/'FREEZE.json')=='b0521e9e2cd30a86ffd366ede1b19ec9a0f06434ac5aca4b73d7611f1f7e4e91'
old_freeze=json.loads((source/'FREEZE.json').read_text())
raw=json.loads(inventory_path.read_text());raw=raw.get('files',raw);inventory={n:(v['sha256']if isinstance(v,dict)else v)for n,v in raw.items()}
files=[p for p in artifact.rglob('*')if p.is_file()]
assert set(inventory)=={str(p.relative_to(artifact))for p in files}
for p in artifact.rglob('*'):assert not p.is_symlink()
for n,h in inventory.items():assert sha(artifact/n)==h,n
for n in ['index.d.cts','index.d.mts']:
 assert sha(artifact/n)==old_freeze['files']['artifact/'+n]['sha256'],n
spec=json.loads((source/'CASE-SPEC.json').read_text());spec['negatives']=[]
inputs=['qualify-portable.cjs','CASE-SPEC.json']
for kind in ['cjs','esm','mixed']:
 old=source/('consumer-'+kind);new=base/('consumer-'+kind);new.mkdir()
 shutil.copytree(artifact,new/'node_modules/@freedom/restricted-railgun-node-prototype')
 for n in ['package.json','mixed.mts'if kind=='mixed'else 'positive.cts'if kind=='cjs'else 'positive.mts']:
  relative=str((old/n).relative_to(source));assert sha(old/n)==old_freeze['files'][relative]['sha256'];shutil.copyfile(old/n,new/n);inputs.append(relative)
 for n,h in inventory.items():assert sha(new/'node_modules/@freedom/restricted-railgun-node-prototype'/n)==h,n
assert sha(source/'qualify-portable.cjs')==old_freeze['files']['qualify-portable.cjs']['sha256']
shutil.copyfile(source/'qualify-portable.cjs',base/'qualify-portable.cjs')
(base/'CASE-SPEC.json').write_text(json.dumps(spec,indent=2)+'\n')
report={'artifactPath':str(artifact),'artifactInventoryPath':str(inventory_path),'artifactInventorySha256':expected_inventory_sha,'artifactInventory':inventory,'historicalTypeFreezeSha256':sha(source/'FREEZE.json'),'positiveCaseSourcePins':{n:sha(source/n)for n in inputs},'declarationsByteUnchanged':True,'physicalCopiesVerified':3,'harnessUnchanged':True,'negativeCasesRepeated':False,'upstreamSourceGraphRepeated':False,'sourceGraphExecuted':False,'preparationRecipeSha256':sha(pathlib.Path(__file__))}
(base/'PREPARE-REPORT.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'prepared':True,'prepareReportSha256':sha(base/'PREPARE-REPORT.json'),'artifactFiles':len(inventory),'positiveCases':3}))

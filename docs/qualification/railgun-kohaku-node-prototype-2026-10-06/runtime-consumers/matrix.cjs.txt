/** Public-vector checks of emitted runtime; independent trusted hosts only. */
const assert = require('assert/strict');
const fixture = require('./independent-host.cjs');
const { independentPrivateHost, deferred, INSTANCE, HASH } = fixture;
const CODE = 'RAILGUN_KOHAKU_PRIVATE_ADAPTER_REFUSED';
const tick = () => new Promise((resolve) => setImmediate(resolve));
const vectors = [
  ['lower', '0x8ba1f109551bd432803012645ac136ddd64dba72', true],
  ['upper', '0x8BA1F109551BD432803012645AC136DDD64DBA72', true],
  ['checksum', '0x8ba1f109551bD432803012645Ac136ddd64DBA72', true],
  ['icap', 'XE65GB6LDNXYOFTX0NSV3FUWKOWIXAMJK36', false],
  ['unprefixed', '8ba1f109551bd432803012645ac136ddd64dba72', false],
  ['bad-checksum', '0x8Ba1f109551bD432803012645Ac136ddd64DBA72', false],
  ['short', '0x1234', false],
  ['nonhex', '0x8ba1f109551bd432803012645ac136ddd64dba7g', false],
];
async function runMatrix(api, other, clean) {
  const results = [],
    rejected = [];
  const onUnhandled = (error) => rejected.push(error);
  process.on('unhandledRejection', onUnhandled);
  async function test(name, action) {
    const opened = [],
      releases = [];
    function open(change = () => {}) {
      const f = independentPrivateHost();
      change(f);
      const adapter = api.createRailgunKohakuPrivateAdapter({
        host: f.host,
        signal: new AbortController().signal,
      });
      const broadcaster = api.createRailgunKohakuPrivateAdapterBroadcaster(adapter);
      opened.push(adapter);
      return {
        ...f,
        adapter,
        broadcaster,
        prepare: () => adapter.prepareTransfer(f.input, INSTANCE),
      };
    }
    let timer;
    try {
      await Promise.race([
        action(open, releases),
        new Promise((_, reject) => {
          timer = setTimeout(() => reject(Error('Case timeout: ' + name)), 3000);
        }),
      ]);
      await tick();
      assert.deepEqual(rejected, []);
      clean();
      results.push(name);
    } finally {
      clearTimeout(timer);
      for (const release of releases) release();
      for (const adapter of opened) adapter.close();
      let cleanupTimer;
      try {
        await Promise.race([
          Promise.allSettled(opened.map((adapter) => adapter.closed)),
          new Promise((_, reject) => {
            cleanupTimer = setTimeout(() => reject(Error('Cleanup timeout: ' + name)), 3000);
          }),
        ]);
      } finally {
        clearTimeout(cleanupTimer);
      }
    }
  }
  try {
    for (const field of ['from', 'to']) {
      for (const [name, address, valid] of vectors) {
        await test(field + ':' + name, async (open) => {
          const f = open((f) => {
            f.acknowledged[field] = address;
          });
          const token = await f.prepare();
          const result = await f.broadcaster.broadcast(token);
          if (valid) {
            assert.equal(result, f.acknowledged);
            assert.equal(result[field], address);
          } else {
            assert.deepEqual(result, { transactionHash: HASH, submissionStatus: 'unknown' });
            assert.notEqual(result, f.acknowledged);
            assert.equal(Object.isFrozen(result), true);
          }
          await f.adapter.closed;
          await assert.rejects(f.broadcaster.broadcast(token), { code: CODE });
          assert.equal(f.calls.filter((c) => c.method === 'broadcast').length, 1);
          assert.equal(f.calls.filter((c) => c.method === 'close').length, 1);
        });
      }
    }
    for (const stop of ['close', 'host-abort']) {
      await test('held-preparation:' + stop, async (open, releases) => {
        const held = deferred(),
          drain = deferred();
        releases.push(
          () => held.resolve({ handle: Object.freeze({}) }),
          () => drain.resolve()
        );
        let closes = 0;
        const f = open((f) => {
          f.host.prepareTransfer = () => held.promise;
          f.host.closed = drain.promise;
          f.host.close = () => {
            closes++;
          };
        });
        const pending = f.prepare();
        let closed = false;
        f.adapter.closed.then(() => {
          closed = true;
        });
        if (stop === 'close') f.adapter.close();
        else f.controller.abort();
        await assert.rejects(pending, { code: CODE });
        await tick();
        assert.equal(closed, false);
        held.resolve({ handle: Object.freeze({}) });
        await tick();
        assert.equal(closed, false);
        drain.resolve();
        await f.adapter.closed;
        assert.equal(closes, 1);
        assert.equal(f.calls.filter((c) => c.method === 'broadcast').length, 0);
      });
    }
    for (const result of [
      { status: 'recovery-required', stage: 'review-draining' },
      { transactionHash: HASH, submissionStatus: 'unknown' },
    ]) {
      await test(
        'held-broadcast:' + (result.stage || result.submissionStatus),
        async (open, releases) => {
          const held = deferred(),
            drain = deferred();
          releases.push(
            () => held.resolve(result),
            () => drain.resolve()
          );
          let calls = 0,
            closed = false,
            settled = false;
          const f = open((f) => {
            f.host.broadcast = () => {
              calls++;
              return held.promise;
            };
            f.host.closed = drain.promise;
            f.host.close = () => {};
          });
          const token = await f.prepare();
          const pending = f.broadcaster.broadcast(token);
          pending.then(() => {
            settled = true;
          });
          f.adapter.closed.then(() => {
            closed = true;
          });
          f.adapter.close();
          await assert.rejects(f.broadcaster.broadcast(token), { code: CODE });
          await tick();
          assert.equal(settled, false);
          assert.equal(closed, false);
          held.resolve(result);
          assert.equal(await pending, result);
          assert.equal(closed, false);
          drain.resolve();
          await f.adapter.closed;
          assert.equal(calls, 1);
        }
      );
    }
    await test('host-closed-first', async (open) => {
      const f = open();
      f.drain.resolve();
      await f.adapter.closed;
      await assert.rejects(f.adapter.notes(), { code: CODE });
      await assert.rejects(f.prepare(), { code: CODE });
      assert.equal(f.calls.filter((c) => c.method === 'close').length, 1);
    });
    for (const kind of ['throw', 'rejection', 'decorated-rejection', 'thenable', 'bad-result']) {
      await test('malformed-preparation:' + kind, async (open) => {
        let called = 0;
        const f = open((f) => {
          f.host.prepareTransfer = () => {
            if (kind === 'throw') throw Error('private host text');
            if (kind === 'rejection') return Promise.reject(Error('private host text'));
            if (kind === 'decorated-rejection')
              return Object.assign(Promise.reject(Error('private host text')), { extra: true });
            if (kind === 'thenable')
              return {
                then() {
                  called++;
                },
              };
            return Promise.resolve({ handle: {} });
          };
        });
        await assert.rejects(f.prepare(), {
          code: CODE,
          message: 'Kohaku private adapter unavailable',
        });
        await f.adapter.closed;
        assert.equal(called, 0);
        await assert.rejects(f.prepare(), { code: CODE });
      });
    }
    await test('late-then-getter', async (open) => {
      let getter = 0;
      const f = open((f) => {
        f.host.broadcast = () => {
          const result = { ...f.acknowledged };
          const pending = Promise.resolve(result);
          Object.defineProperty(result, 'then', {
            get() {
              getter++;
              throw Error('then invoked');
            },
          });
          return pending;
        };
      });
      assert.deepEqual(await f.broadcaster.broadcast(await f.prepare()), {
        transactionHash: HASH,
        submissionStatus: 'unknown',
      });
      await f.adapter.closed;
      assert.equal(getter, 0);
    });
    for (const kind of ['unique', 'conflict', 'getter']) {
      await test('malformed-outcome:' + kind, async (open) => {
        let getter = 0;
        const value =
          kind === 'unique'
            ? { hash: HASH, malformed: true }
            : kind === 'conflict'
              ? { hash: HASH, transactionHash: HASH }
              : Object.defineProperty({}, 'hash', {
                  get() {
                    getter++;
                    return HASH;
                  },
                });
        const f = open((f) => {
          f.host.broadcast = async () => value;
        });
        assert.deepEqual(
          await f.broadcaster.broadcast(await f.prepare()),
          kind === 'unique'
            ? { transactionHash: HASH, submissionStatus: 'unknown' }
            : { status: 'recovery-required', stage: 'adapter-contract' }
        );
        await f.adapter.closed;
        assert.equal(getter, 0);
      });
    }
    await test('invalid-input-then-valid-and-copy-replay', async (open) => {
      const f = open();
      await assert.rejects(f.adapter.prepareTransfer({ ...f.input, amount: 1 }, INSTANCE), {
        code: CODE,
      });
      assert.equal(f.calls.length, 0);
      const token = await f.prepare();
      await assert.rejects(f.broadcaster.broadcast({ ...token }), { code: CODE });
      assert.equal(await f.broadcaster.broadcast(token), f.acknowledged);
      await f.adapter.closed;
      await assert.rejects(f.broadcaster.broadcast(token), { code: CODE });
      assert.equal(f.calls.filter((c) => c.method === 'broadcast').length, 1);
    });
    await test('close-invalidates-unused-token', async (open) => {
      const f = open(),
        token = await f.prepare();
      f.adapter.close();
      await f.adapter.closed;
      await assert.rejects(f.broadcaster.broadcast(token), { code: CODE });
      assert.equal(f.calls.filter((c) => c.method === 'broadcast').length, 0);
    });
    await test('detached-reads-and-invalid-read', async (open) => {
      const f = open();
      const notes = await f.adapter.notes();
      notes[0].amount = 19n;
      notes[0].asset.contract = '0x' + 'ff'.repeat(20);
      notes.push({});
      assert.equal((await f.adapter.notes())[0].amount, 2000n);
      assert.notEqual((await f.adapter.notes())[0].asset.contract, notes[0].asset.contract);
      f.values[0].amount = 2;
      await assert.rejects(f.adapter.notes(), { code: CODE });
      f.adapter.close();
      await f.adapter.closed;
    });
    if (other)
      await test('physical-copy-token-isolation', async (open) => {
        assert.notEqual(
          api.createRailgunKohakuPrivateAdapter,
          other.createRailgunKohakuPrivateAdapter
        );
        const a = open(),
          g = independentPrivateHost();
        const b = other.createRailgunKohakuPrivateAdapter({
          host: g.host,
          signal: new AbortController().signal,
        });
        try {
          const tx = other.createRailgunKohakuPrivateAdapterBroadcaster(b);
          assert.throws(() => other.createRailgunKohakuPrivateAdapterBroadcaster(a.adapter), {
            code: CODE,
          });
          const one = await a.prepare(),
            two = await b.prepareTransfer(g.input, INSTANCE);
          await assert.rejects(tx.broadcast(one), { code: CODE });
          await assert.rejects(a.broadcaster.broadcast(two), { code: CODE });
          assert.equal(a.calls.filter((c) => c.method === 'broadcast').length, 0);
          assert.equal(g.calls.filter((c) => c.method === 'broadcast').length, 0);
          assert.equal(await a.broadcaster.broadcast(one), a.acknowledged);
          assert.equal(await tx.broadcast(two), g.acknowledged);
          await Promise.all([a.adapter.closed, b.closed]);
          await assert.rejects(tx.broadcast(two), { code: CODE });
          await assert.rejects(a.broadcaster.broadcast(one), { code: CODE });
          assert.equal(g.calls.filter((c) => c.method === 'broadcast').length, 1);
        } finally {
          b.close();
          await b.closed;
        }
      });
    return {
      cases: results,
      count: results.length,
      addressVectors: vectors.length,
      addressFields: 2,
      unhandledRejections: rejected.length,
      sharedHostAcrossCopiesSupported: false,
    };
  } finally {
    process.removeListener('unhandledRejection', onUnhandled);
  }
}
module.exports = { runMatrix };

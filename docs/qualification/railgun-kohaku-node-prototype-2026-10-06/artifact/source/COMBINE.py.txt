"""Deterministic scratch combiner; no dependency installation or runtime execution."""
import hashlib
import json
import os
from pathlib import Path
import re
import shutil
import sys

if not __debug__:
    raise RuntimeError('Optimized Python is unsupported')
if len(sys.argv) != 4:
    raise RuntimeError('Usage: python3 combine.py RUNTIME_LANE TYPE_LANE FRESH_OUTPUT')
runtime, types, output = map(lambda value: Path(value).absolute(), sys.argv[1:])
assert not os.path.lexists(output), 'Output must be fresh'
assert output.parent.resolve() == output.parent
assert str(output).startswith('<SCRATCH_ROOT>/')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def regular(path):
    assert path.is_file() and not path.is_symlink() and path.resolve() == path
    return path

def inventory(directory):
    result = {}
    for path in sorted(directory.rglob('*')):
        assert not path.is_symlink()
        if path.is_file():
            result[str(path.relative_to(directory))] = sha(regular(path))
        else:
            assert path.is_dir()
    return result

def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    assert not path.exists()
    path.write_text(json.dumps(value, indent=2) + '\n')

def copy(source, destination):
    regular(source)
    destination.parent.mkdir(parents=True, exist_ok=True)
    assert not destination.exists()
    shutil.copyfile(source, destination)

assert sha(regular(runtime / 'FREEZE.json')) == 'be83ace0d7c102bfbcc5c78eb2791598ec94aa4470b4c30b2a65d0b4cdf21688'
assert sha(regular(types / 'FREEZE.json')) == 'b0521e9e2cd30a86ffd366ede1b19ec9a0f06434ac5aca4b73d7611f1f7e4e91'
runtime_freeze = json.loads((runtime / 'FREEZE.json').read_text())
type_freeze = json.loads((types / 'FREEZE.json').read_text())
source = runtime / 'build-b/artifact'
source_types = types / 'artifact'
runtime_inventory = inventory(source)
type_inventory = inventory(source_types)
assert runtime_inventory == {k[len('build-b/artifact/'):]: v for k, v in runtime_freeze['files'].items() if k.startswith('build-b/artifact/')}
assert type_inventory == {k[len('artifact/'):]: v['sha256'] for k, v in type_freeze['files'].items() if k.startswith('artifact/')}
for lane, freeze, name in [(runtime, runtime_freeze, 'build-b/BUILD-REPORT.json'), (types, type_freeze, 'prepare.py')]:
    expected = freeze['files'][name]
    if isinstance(expected, dict):
        expected = expected['sha256']
    assert sha(regular(lane / name)) == expected

# Recreate the reviewed declaration transformation using exact original contracts.
snapshot_path = 'source/scripts/fixtures/railgun-kohaku-snapshot-contract.d.ts'
private_path = 'source/scripts/fixtures/railgun-kohaku-private-contract.d.ts'
snapshot = (source_types / snapshot_path).read_text()
snapshot = snapshot[snapshot.index('export type ReadAsset ='):snapshot.index('// The actual pinned target,')].rstrip() + '\n'
private = (source_types / private_path).read_text()
private = private[private.index('export interface PrivateInput {'):]
line = 'export type PinnedPrivateTarget = PluginInstance<string, PrivateCapabilities>;\n'
assert private.count(line) == 1
private = private.replace(line, '')
line = 'export type PrivateAdapterBroadcaster = Broadcaster<PrivateOperation, PrivateSubmissionOutcome>;'
assert private.count(line) == 1
private = private.replace(line, 'export type PrivateAdapterBroadcaster = {\n  broadcast(operation: PrivateOperation): Promise<PrivateSubmissionOutcome>;\n};')
canonical = '/** Freedom restricted trusted-host declarations, derived from pinned MPL-2.0 sources.\n * Host authority, native Promise and identity checks remain runtime contracts.\n * This file owns the only distribution PrivateOperation brand.\n */\n' + snapshot + '\n' + private
assert not re.search(r'\bimport\b|\bPluginInstance\b|\bPinned\w*Target\b|Broadcaster<|declare\s+module', canonical)
names = re.findall(r'export (?:type|interface) (\w+)', canonical)
functions = re.findall(r'export declare function (\w+)', canonical)
assert len(functions) == 3 and len(names) == len(set(names))
esm = '/** Forward both entrypoints to one declaration identity. */\nexport {\n' + ''.join('  ' + name + ',\n' for name in functions) + "} from './index.cjs';\nexport type {\n" + ''.join('  ' + name + ',\n' for name in names) + "} from './index.cjs';\n"
assert canonical.encode() == (source_types / 'index.d.cts').read_bytes()
assert esm.encode() == (source_types / 'index.d.mts').read_bytes()
manifest = json.loads((source / 'package.json').read_text())
manifest['types'] = './index.d.cts'
manifest['exports']['.'] = {'import': {'types': './index.d.mts', 'default': './index.mjs'}, 'require': {'types': './index.d.cts', 'default': './index.cjs'}}
assert (json.dumps(manifest, indent=2) + '\n').encode() == (source_types / 'package.json').read_bytes()

output.mkdir()
artifact = output / 'artifact'
artifact.mkdir()
for name in runtime_inventory:
    if name != 'package.json':
        copy(source / name, artifact / name)
for name in ['package.json', 'index.d.cts', 'index.d.mts', snapshot_path, private_path]:
    copy(source_types / name, artifact / name)
copy(types / 'prepare.py', artifact / 'provenance/historical-type-lane-prepare.py')
copy(source_types / 'TYPE-SOURCE-BASIS.json', artifact / 'provenance/historical-type-lane-source-basis.json')
copy(runtime / 'build-b/BUILD-REPORT.json', artifact / 'provenance/runtime-phase-build-report.json')
copy(regular(Path(__file__).absolute()), artifact / 'source/COMBINE.py')
recipe_hash = sha(Path(__file__).absolute())
write(artifact / 'TYPE-SOURCE-BASIS.json', {
    'schema': 'restricted-node-combined-type-basis-v1',
    'runtimeSha256': sha(artifact / 'runtime.cjs'),
    'packageSha256': sha(artifact / 'package.json'),
    'canonicalDeclarationSha256': sha(artifact / 'index.d.cts'),
    'forwardingDeclarationSha256': sha(artifact / 'index.d.mts'),
    'typeLaneFreezeSha256': sha(types / 'FREEZE.json'),
    'historicalTypeBasisSha256': sha(source_types / 'TYPE-SOURCE-BASIS.json'),
    'historicalTypeRecipeSha256': sha(types / 'prepare.py'),
    'combinationRecipeSha256': recipe_hash,
    'originalContracts': {name: sha(artifact / name) for name in [snapshot_path, private_path]},
    'declarationTransformReproducedExactly': True,
    'historicalTypeRuntimeIsNotFinalRuntime': True,
    'finalPackageCompilerQualification': 'separate observation required',
})
(artifact / 'REPRODUCE.md').write_text('''# Combined restricted Node prototype

This unpublished artifact joins the corrected phase-two runtime with the reviewed portable declarations. `source/COMBINE.py` verifies the two frozen lane inventories, reproduces the exact declaration/manifest transform, and assembles a fresh output deterministically. Run `python3 combine.py RUNTIME_LANE TYPE_LANE FRESH_OUTPUT` using preserved reviewed lanes. There is no installation, runtime execution or source compilation in that assembly step.

The combiner includes the effective declaration transform without requiring execution of that historical hardcoded recipe. `TYPE-SOURCE-BASIS.json` records the corrected final runtime, manifest and unchanged declaration joins. Existing full license texts and exact Freedom runtime sources are retained. The two original Freedom MPL-2.0 contract sources are included under source/scripts/fixtures; their derived declarations remain covered by licenses/freedom/LICENSE. There are no new bundled third-party dependencies or upstream source files. Build tools and compiler are not shipped runtime dependencies. This inventory is not distribution approval.

Use one physical package for both CJS and ESM so identity maps and the type-only operation brand share their canonical modules. Same host across independent copies/realms is unsupported. final = inventory(artifact)
assert inventory(source) == runtime_inventory
assert inventory(source_types) == type_inventory
for name in runtime_inventory:
    if name != 'package.json':
        assert final[name] == runtime_inventory[name]
write(output / 'ARTIFACT-INVENTORY.json', final)
write(output / 'COMBINED-REPORT.json', {
    'schema': 'restricted-node-combination-v1',
    'runtimeLaneFreezeSha256': sha(runtime / 'FREEZE.json'),
    'typeLaneFreezeSha256': sha(types / 'FREEZE.json'),
    'recipeSha256': recipe_hash,
    'runtimeInputInventory': runtime_inventory,
    'typeInputInventory': type_inventory,
    'runtimeUnchanged': True,
    'declarationsUnchanged': True,
    'declarationTransformReproducedExactly': True,
    'manifestTransition': {'runtimePhaseSha256': runtime_inventory['package.json'], 'finalSha256': final['package.json']},
    'finalArtifactInventorySha256': sha(output / 'ARTIFACT-INVENTORY.json'),
    'finalFiles': len(final),
    'runtimeOrCompilerExecutedDuringCombination': False,
})
print(json.dumps({'status': 'combined', 'files': len(final), 'inventorySha256': sha(output / 'ARTIFACT-INVENTORY.json')}))

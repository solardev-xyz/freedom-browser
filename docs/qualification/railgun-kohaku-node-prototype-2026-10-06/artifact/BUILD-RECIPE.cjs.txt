/** Scratch-only deterministic artifact build. Never imports a Freedom module. */
const assert = require('assert/strict');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');
assert.equal(process.argv.length, 5, 'Usage: node build.cjs ROOT OUTPUT SOURCE-BASIS.json');
const ROOT = path.resolve(process.argv[2]);
const basisFilename = path.resolve(process.argv[4]);
const BASIS = JSON.parse(fs.readFileSync(basisFilename));
const esbuild = require(ROOT + '/node_modules/esbuild');
const parser = require(ROOT + '/node_modules/@babel/parser');
const sha = (bytes) => crypto.createHash('sha256').update(bytes).digest('hex');
const relative = (file) => path.relative(ROOT, file).split(path.sep).join('/');
const authored = [
  'src/main/wallet/railgun-kohaku-snapshot-plugin.js',
  'src/main/wallet/railgun-kohaku-private-adapter.js',
  'src/main/wallet/railgun-kohaku-read-data.js',
  'src/main/wallet/railgun-kohaku-read-dispatch.js',
  'src/main/wallet/railgun-shield-pins.json',
];
const inputBytes = new Map();
function read(file) {
  const resolved = path.resolve(file);
  assert.equal(fs.realpathSync(resolved), resolved, 'Input realpath/symlink mismatch');
  assert.ok(fs.lstatSync(resolved).isFile());
  const bytes = fs.readFileSync(resolved);
  if (inputBytes.has(resolved)) assert.deepEqual(bytes, inputBytes.get(resolved), 'Input changed');
  else inputBytes.set(resolved, bytes);
  return bytes;
}
function json(file) {
  return JSON.parse(read(file));
}
function recordMetadata(file) {
  let dir = path.dirname(file);
  while (dir === ROOT || dir.startsWith(ROOT + path.sep)) {
    const candidate = path.join(dir, 'package.json');
    if (fs.existsSync(candidate)) read(candidate);
    if (dir === ROOT) break;
    dir = path.dirname(dir);
  }
}
const ENTRY = `const snapshot = require('./src/main/wallet/railgun-kohaku-snapshot-plugin.js');
const privateAdapter = require('./src/main/wallet/railgun-kohaku-private-adapter.js');
module.exports = Object.freeze({
  createRailgunKohakuSnapshotPlugin: snapshot.createRailgunKohakuSnapshotPlugin,
  createRailgunKohakuPrivateAdapter: privateAdapter.createRailgunKohakuPrivateAdapter,
  createRailgunKohakuPrivateAdapterBroadcaster: privateAdapter.createRailgunKohakuPrivateAdapterBroadcaster,
});\n`;
const BRIDGE = `export { getAddress } from 'ethers/address';\n`;
const CJS = `module.exports = require('./runtime.cjs');\n`;
const ESM = `import runtime from './runtime.cjs';
export const createRailgunKohakuSnapshotPlugin = runtime.createRailgunKohakuSnapshotPlugin;
export const createRailgunKohakuPrivateAdapter = runtime.createRailgunKohakuPrivateAdapter;
export const createRailgunKohakuPrivateAdapterBroadcaster = runtime.createRailgunKohakuPrivateAdapterBroadcaster;\n`;
function packageOwner(file) {
  let dir = path.dirname(file);
  while (dir.startsWith(ROOT + path.sep)) {
    const candidate = path.join(dir, 'package.json');
    if (fs.existsSync(candidate)) {
      const info = json(candidate);
      if (typeof info.name === 'string') return { dir, info };
    }
    dir = path.dirname(dir);
  }
  return { dir: ROOT, info: json(ROOT + '/package.json') };
}
function auditOutput(result) {
  const output = result.metafile.outputs['runtime.cjs'];
  assert.ok(output);
  const allowed = new Set(['assert/strict', 'util']);
  for (const imported of output.imports) {
    assert.equal(imported.external, true);
    assert.ok(allowed.has(imported.path), 'Unexpected runtime external: ' + imported.path);
  }
  const emitted = Object.entries(output.inputs).filter(([, value]) => value.bytesInOutput > 0);
  for (const [file] of emitted) {
    assert.ok(
      !/\/(providers|wallet|contract)\//.test(file) || authored.includes(file),
      'Forbidden contributor ' + file
    );
    assert.ok(
      !/\/(fetch|geturl|provider-[^/]*|ws)\.js$|node_modules\/(ws|electron)\//.test(file),
      'Network contributor ' + file
    );
    if (file.startsWith('src/'))
      assert.ok(authored.includes(file), 'Freedom authority entered artifact');
  }
  const ast = parser.parse(result.outputFiles[0].text, { sourceType: 'script' });
  let requireCalls = 0,
    commonJSWrapperNames = 0;
  function walk(node, parent) {
    if (!node || typeof node !== 'object') return;
    assert.ok(
      !['ImportDeclaration', 'ImportExpression'].includes(node.type),
      'Unexpected dynamic/module import'
    );
    if (
      node.type === 'CallExpression' &&
      node.callee.type === 'Identifier' &&
      node.callee.name === 'require'
    ) {
      assert.equal(node.arguments.length, 1);
      assert.equal(node.arguments[0].type, 'StringLiteral');
      assert.ok(allowed.has(node.arguments[0].value), 'Unadmitted require');
      requireCalls++;
    }
    if (node.type === 'Identifier' && node.name === '__require') {
      assert.equal(parent?.type, 'FunctionExpression');
      assert.equal(parent.id, node);
      assert.equal(parent.params.length, 0);
      commonJSWrapperNames++;
    }
    if (node.type === 'Identifier')
      assert.ok(
        !['fetch', 'WebSocket', 'XMLHttpRequest', 'Worker', 'process', 'Electron'].includes(
          node.name
        ),
        'Forbidden runtime identifier ' + node.name
      );
    for (const value of Object.values(node)) {
      if (Array.isArray(value)) value.forEach((child) => walk(child, node));
      else if (value && typeof value === 'object') walk(value, node);
    }
  }
  walk(ast);
  assert.equal(commonJSWrapperNames, 1);
  return {
    parsedInputs: Object.keys(result.metafile.inputs).length,
    emittedInputs: emitted.map(([file, value]) => ({ file, ...value })),
    externalImports: output.imports,
    requireCalls,
    commonJSWrapperNames,
    outputBytes: output.bytes,
  };
}
async function main() {
  assert.equal(process.argv.length, 5);
  const destination = path.resolve(process.argv[3]);
  assert.ok(destination.startsWith('<SCRATCH_ROOT>/'));
  assert.ok(!fs.existsSync(destination) && !fs.lstatSync(destination, { throwIfNoEntry: false }));
  assert.ok(!process.env.NODE_OPTIONS && !process.env.NODE_PATH, 'Unreviewed Node preload/path');
  assert.equal(process.version, 'v24.18.1');
  assert.equal(esbuild.version, '0.28.2');
  assert.equal(process.platform, 'darwin');
  assert.equal(process.arch, 'arm64');
  assert.ok(!process.env.ESBUILD_BINARY_PATH, 'Unreviewed esbuild override');
  for (const name of authored) assert.equal(sha(read(ROOT + '/' + name)), BASIS.sources[name]);
  for (const name of [
    'package.json',
    'package-lock.json',
    'LICENSE',
    'node_modules/esbuild/package.json',
    'node_modules/ethers/package.json',
  ])
    assert.equal(sha(read(ROOT + '/' + name)), BASIS.sources[name]);
  const toolPackage = json(ROOT + '/node_modules/esbuild/package.json');
  const binary = ROOT + '/node_modules/@esbuild/darwin-arm64/bin/esbuild';
  assert.equal(
    sha(read(binary)),
    toolPackage['esbuild.binaryHashes']['@esbuild/darwin-arm64/bin/esbuild']
  );
  read(ROOT + '/node_modules/@esbuild/darwin-arm64/package.json');
  read(ROOT + '/node_modules/esbuild/lib/main.js');
  read(ROOT + '/node_modules/@babel/parser/lib/index.js');
  read(ROOT + '/node_modules/@babel/parser/package.json');
  const nodeBinary = fs.realpathSync(process.execPath),
    nodeSha256 = sha(read(nodeBinary));
  let bridgeResolutions = 0;
  const result = await esbuild.build({
    absWorkingDir: ROOT,
    stdin: { contents: ENTRY, resolveDir: ROOT, sourcefile: 'restricted-entry.cjs', loader: 'js' },
    outfile: 'runtime.cjs',
    bundle: true,
    write: false,
    metafile: true,
    platform: 'node',
    format: 'cjs',
    target: 'node24',
    treeShaking: true,
    minify: false,
    sourcemap: false,
    legalComments: 'inline',
    charset: 'utf8',
    mainFields: ['module', 'main'],
    conditions: ['node'],
    tsconfigRaw: {},
    plugins: [
      {
        name: 'restricted-ethers-import',
        setup(build) {
          build.onResolve({ filter: /^ethers$/ }, (args) => {
            assert.equal(
              args.importer,
              ROOT + '/src/main/wallet/railgun-kohaku-private-adapter.js'
            );
            assert.equal(args.kind, 'require-call');
            bridgeResolutions++;
            return { path: 'ethers-address-only', namespace: 'restricted-bridge' };
          });
          build.onLoad({ filter: /.*/, namespace: 'restricted-bridge' }, () => ({
            contents: BRIDGE,
            loader: 'js',
            resolveDir: ROOT,
          }));
          build.onLoad({ filter: /.*/, namespace: 'file' }, (args) => {
            assert.ok(
              args.path.startsWith(ROOT + '/node_modules/') ||
                authored.includes(relative(args.path)),
              'Unexpected source'
            );
            recordMetadata(args.path);
            return {
              contents: read(args.path),
              loader: args.path.endsWith('.json') ? 'json' : 'js',
            };
          });
        },
      },
    ],
  });
  assert.equal(bridgeResolutions, 1);
  assert.equal(result.errors.length, 0);
  assert.equal(result.warnings.length, 0);
  fs.mkdirSync(destination, { mode: 0o700 });
  fs.writeFileSync(
    path.join(destination, 'metafile.json'),
    JSON.stringify(result.metafile, null, 2) + '\n'
  );
  fs.writeFileSync(path.join(destination, 'runtime-output.cjs'), result.outputFiles[0].contents);
  const graph = auditOutput(result);
  const packages = new Map();
  for (const file of Object.keys(result.metafile.inputs)) {
    if (!file.startsWith('node_modules/')) continue;
    const owner = packageOwner(ROOT + '/' + file);
    if (!packages.has(owner.dir)) packages.set(owner.dir, owner);
  }
  const emittedPaths = new Set(graph.emittedInputs.map((v) => v.file));
  const licenseRows = [];
  const artifact = path.join(destination, 'artifact');
  fs.mkdirSync(artifact);
  const put = (name, bytes) => {
    const out = path.join(artifact, name);
    fs.mkdirSync(path.dirname(out), { recursive: true });
    fs.writeFileSync(out, bytes);
  };
  for (const owner of [{ dir: ROOT, info: json(ROOT + '/package.json') }, ...packages.values()]) {
    const licenses = fs
      .readdirSync(owner.dir)
      .filter((name) => /^(licen[cs]e|copying|notice)(\.|$)/i.test(name));
    assert.ok(licenses.length > 0, 'No license: ' + owner.info.name);
    const label = owner.dir === ROOT ? 'freedom' : relative(owner.dir).replaceAll('/', '__');
    const shipped =
      owner.dir === ROOT ||
      [...emittedPaths].some((file) => file.startsWith(relative(owner.dir) + '/'));
    for (const name of licenses) {
      const bytes = read(path.join(owner.dir, name));
      put('licenses/' + label + '/' + name, bytes);
      licenseRows.push({
        package: owner.info.name,
        version: owner.info.version,
        packageRoot: owner.dir === ROOT ? '.' : relative(owner.dir),
        declaredLicense: owner.info.license,
        emittedContributor: shipped,
        file: 'licenses/' + label + '/' + name,
        sha256: sha(bytes),
      });
    }
  }
  const buildTools = [];
  for (const name of ['esbuild', '@babel/parser']) {
    const dir = ROOT + '/node_modules/' + name,
      info = json(dir + '/package.json');
    const license = fs.readdirSync(dir).filter((name) => /^license(\.|$)/i.test(name));
    assert.equal(license.length, 1);
    const bytes = read(path.join(dir, license[0]));
    put('build-tool-licenses/' + name.replaceAll('/', '__') + '/' + license[0], bytes);
    buildTools.push({
      name,
      version: info.version,
      license: license[0],
      sha256: sha(bytes),
      runtimeDependency: false,
    });
  }
  put('runtime.cjs', result.outputFiles[0].contents);
  put('index.cjs', CJS);
  put('index.mjs', ESM);
  put(
    'package.json',
    JSON.stringify(
      {
        name: '@freedom/restricted-railgun-node-prototype',
        version: '0.0.0-private',
        private: true,
        license: 'MPL-2.0',
        main: './index.cjs',
        exports: { '.': { import: './index.mjs', require: './index.cjs' } },
        engines: { node: '24.18.1' },
      },
      null,
      2
    ) + '\n'
  );
  for (const name of authored) put('source/' + name, read(ROOT + '/' + name));
  put('source/restricted-entry.cjs', ENTRY);
  put('source/ethers-address-only.mjs', BRIDGE);
  put('BUILD-RECIPE.cjs', fs.readFileSync(__filename));
  put('source/SOURCE-BASIS.json', fs.readFileSync(basisFilename));
  put(
    'LICENSE-INVENTORY.json',
    JSON.stringify(
      licenseRows.sort((a, b) => a.file.localeCompare(b.file)),
      null,
      2
    ) + '\n'
  );
  const inventory = {};
  for (const [file, bytes] of [...inputBytes].sort(([a], [b]) => a.localeCompare(b))) {
    assert.equal(fs.realpathSync(file), file, 'Input path changed by end');
    assert.ok(fs.lstatSync(file).isFile());
    assert.deepEqual(fs.readFileSync(file), bytes, 'Input changed by end');
    inventory[file === nodeBinary ? '<node-executable>' : relative(file)] = {
      sha256: sha(bytes),
      bytes: bytes.length,
    };
  }
  const graphReport = {
    ...graph,
    buildTools,
    recipeSha256: sha(fs.readFileSync(__filename)),
    designBasisSha256: sha(fs.readFileSync(basisFilename)),
    bridgeResolutions,
    inputInventory: inventory,
    tools: { node: process.version, nodeSha256, esbuild: esbuild.version },
    licenses: licenseRows,
  };
  fs.writeFileSync(
    path.join(destination, 'BUILD-REPORT.json'),
    JSON.stringify(graphReport, null, 2) + '\n'
  );
  console.log(
    JSON.stringify({
      status: 'built',
      parsedInputs: graph.parsedInputs,
      emittedInputs: graph.emittedInputs.length,
      outputBytes: graph.outputBytes,
      externalImports: graph.externalImports,
      packages: licenseRows.map((v) => v.package + '@' + v.version),
    })
  );
}
main()
  .catch((error) => {
    console.error(error.stack);
    process.exitCode = 1;
  })
  .finally(() => esbuild.stop());
